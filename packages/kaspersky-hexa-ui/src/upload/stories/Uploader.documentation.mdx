                />

          const demoRequest = ({ onProgress, onSuccess }) => {
            let percent = 0
            const timer = setInterval(() => {
              percent += 50
              if (percent < 100) {
                onProgress?.({ percent })
              } else {
                clearInterval(timer)
                onSuccess?.({})
              }
            }, 400)
            return {
              abort() {
                clearInterval(timer)
              },
            }
          }
        `}
