      <DocExampleBlock

          <RichTextEditor
            plugins={allPlugins}
            initialValue={initialValueEmpty}

        `}

      />

      <DocExampleBlock
        title="readOnly и enabled"
        description="**readOnly** включает режим чтения: редактирование и панель форматирования недоступны. **enabled** управляет возможностью изменять текст; по умолчанию — `true`."
        code={`
          <RichTextEditor
            plugins={allPlugins}
            initialValue={initialValue}
            readOnly
            onChange={() => undefined}
          />
        `}
        scope={{ RichTextEditor, allPlugins, initialValue }}
      />

      <DocExampleBlock

            initialValue={longValue}

            onChange={() => undefined}
          />
        `}

      />

      <DocExampleBlock

      <DocExampleBlock

          <RichTextEditor
            plugins={allPlugins}
            initialValue={initialValue}
            limitTextSize={1000}
