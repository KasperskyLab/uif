              if (text.includes('TODO')) {
                return [

                    from: 0,
                    to: text.length,
                    message: 'Remove TODO',
                    severity: 'warning',
                  },
                ]
              }
              return []
            }

            return (
              <CodeViewer
                language="javascript"
                initialValue={'// TODO: fix me'}
                linter={codeViewerLinter}
              />
            )
          }
