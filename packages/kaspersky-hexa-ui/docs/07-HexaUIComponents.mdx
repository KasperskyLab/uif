export const buildStoryId = (path) =>
  path
    .toLowerCase()
    .replace(/[^a-z0-9/ ]/g, '')
    .replace(/[ /]+/g, '-')
    .replace(/-+/g, '-')

export const buildDocsHref = (path, page, view = 'docs') => {
  const storyPath = buildStoryId(path)
  const lastSegment = buildStoryId(path.split('/').pop())
  // New docs pages use `--{name}-docs`; `page` overrides legacy or irregular IDs.
  const docsPage = !page || page === 'documentation' ? `${lastSegment}-docs` : page

  return `/hexa-ui/?path=/${view}/${storyPath}--${docsPage}`
}
